        document.querySelectorAll('.toggle-btn.no').forEach(b => b.innerText = t.noText);

        for (let key in settings) {
            if (settings[key] === 'YES' || settings[key] === 'ДА' || settings[key] === 'TAK') settings[key] = t.yesText;
            else if (settings[key] === 'NO' || settings[key] === 'НЕТ' || settings[key] === 'NIE') settings[key] = t.noText;
            else settings[key] = t.notSet;
        }
        updateCommand();
    }

    function switchTab(evt, tabName) {
        let contents = document.getElementsByClassName("tab-content");
        for (let c of contents) c.classList.remove("active");

        let buttons = document.getElementsByClassName("tab-btn");
        for (let b of buttons) b.classList.remove("active");

        document.getElementById(tabName).classList.add("active");
        evt.currentTarget.classList.add("active");
    }

    function setToggle(button, key, valType) {
        let t = translations[currentLang];
        let actualVal = (valType === 'YES') ? t.yesText : t.noText;
        
        settings[key] = actualVal;
        let parent = button.parentElement;
        let buttons = parent.getElementsByClassName("toggle-btn");
        for (let b of buttons) b.classList.remove("active");
        button.classList.add("active");

        updateCommand();
    }

    function updateCommand() {
        let text = `[SCP CONFIG] SCPs: ${settings.scp} | Strikes: ${settings.strikes} | Cells: ${settings.freedom} | Ranks: ${settings.ranks} | Outside: ${settings.outside}`;
        let command = `/tellraw @a {"text":"${text}","color":"gold"}`;
        document.getElementById("commandOutput").value = command;
    }

    function copyCommand() {
        let copyText = document.getElementById("commandOutput");
        copyText.select();
        document.execCommand("copy");
        alert(translations[currentLang].copiedAlert);
    }
